const fs = require('fs');

fs.readFile('data.txt', 'utf8',
  (err, data) => {
    if (err) throw err;

⚠ Callback Hell : chaque opération async imbrique un nouveau callback.
Propagation d'erreurs manuelle à chaque niveau.
Toujours vérifier err en premier argument (convention Node.js).
ES2015 Promise 
// 3 états : pending → fulfilled | rejected
const p = new Promise((resolve, reject) => {
  // opération asynchrone...
  if (ok) resolve(valeur);
  else   reject(new Error('raison'));
});
// Chaînage
p.then(v => parse(v))
 .then(j => sauvegarder(j))
 .catch(e => console.error(e))
 .finally(() => /* nettoyage */);

// Combinateurs
await Promise.all([p1,p2,p3]);     // toutes OK
await Promise.race([p1,p2]);      // la + rapide
await Promise.allSettled([p1,p2]);// résultats
await Promise.any([p1,p2]);      // 1ère réussie
pending → fulfilled / rejected
.then() retourne une nouvelle Promise → chaînable.
1 seul .catch() en fin de chaîne suffit.
ES2017 async / await 
// Syntaxe sync, sémantique async
async function charger(url) {
  try {
    const rep  = await fetch(url);
    const data = await rep.json();
    return data;
  } catch (err) {
    console.error(err);
  }
}
// Parallèle avec async/await
const [a, b, c] = await Promise.all([
  charger('/api/a'),
  charger('/api/b'),
  charger('/api/c'),
]);
// ⚠ Piège : await séquentiel en boucle
for (const u of urls)
  await charger(u); // ❌ séquentiel !
async → retourne toujours une Promise.
await → suspend la coroutine, cède à l'event loop.
Top-level await disponible en modules ES (ESM).

③ Python — threading vs asyncio vs multiprocessing

# ── threading : parallélisme OS, partage mémoire ─────────
import threading, time

def tache(nom: str):
    time.sleep(1)           # I/O simulé (libère le GIL)
    print(f"[{nom}] terminé")

t1 = threading.Thread(target=tache, args=("A",))
t2 = threading.Thread(target=tache, args=("B",))
t1.start(); t2.start()
t1.join();  t2.join()    # attendre la fin

# ── asyncio : coroutines, 1 thread, event loop ────────────
import asyncio

async def recuperer(url: str) -> str:
    await asyncio.sleep(1)    # ✓ non-bloquant
    return f"données:{url}"

async def main():
    # Lance 3 coroutines en "parallèle"
    resultats = await asyncio.gather(
        recuperer("api/a"),
        recuperer("api/b"),
        recuperer("api/c"),    # ~1s total !
    )

asyncio.run(main())

# ── multiprocessing : vrai parallélisme CPU ───────────────
from multiprocessing import Pool
with Pool(4) as p:              # 4 processus
    res = p.map(calcul, donnees) # contourne le GIL
GIL (Global Interpreter Lock) : 1 seul thread Python actif à la fois sur le CPU.
asyncio.gather() → concurrent (pas vrai //), idéal pour I/O.
CPU-bound → multiprocessing (processus séparés, pas de GIL).

④ Langage C — POSIX Threads (pthreads) · gcc -pthread

/* Compilation : gcc -Wall -pthread -o prog prog.c */
#include <pthread.h>
#include <stdio.h>
#include <stdlib.h>

/* Signature obligatoire : void* f(void* arg) */
void* tache(void* arg) {
    int id = *(int*)arg;          // cast du pointeur
    printf("Thread %d démarre\n", id);
    /* travail ici... */
    return (void*)0;              // valeur de retour
}

int main(void) {
    pthread_t t1, t2;
    int id1 = 1, id2 = 2;

    /* Création — non-bloquant */
    pthread_create(&t1, NULL, tache, &id1);
    pthread_create(&t2, NULL, tache, &id2);

    /* Attente de fin (bloquant pour main) */
    pthread_join(t1, NULL);   // attendre t1
    pthread_join(t2, NULL);   // attendre t2

    printf("Tous les threads terminés\n");
    return 0;
}
pthread_create() : crée un thread OS réel (pas une coroutine).
pthread_join() : attend la fin + récupère la valeur de retour.
Chaque thread a sa propre pile mais partage le tas (heap) et les globaux.

④ Langage C — Mutex, Section critique & Primitives de sync

#include <pthread.h>

/* ── Mutex (exclusion mutuelle) ─────────────── */
pthread_mutex_t verrou = PTHREAD_MUTEX_INITIALIZER;
int compteur = 0;            // ressource partagée

void* incrementer(void* _) {
    for (int i = 0; i < 100000; i++) {
        pthread_mutex_lock(&verrou);   // ← LOCK
        compteur++;                    // section critique
        pthread_mutex_unlock(&verrou); // ← UNLOCK
    }
    return NULL;
}
/* ── Variable condition (signal inter-threads) ─ */
pthread_cond_t  cond  = PTHREAD_COND_INITIALIZER;
pthread_cond_wait(&cond, &verrou);   // attendre signal
pthread_cond_signal(&cond);          // réveiller 1
pthread_cond_broadcast(&cond);       // réveiller tous

/* ── Sémaphore POSIX ─────────────────────────── */
#include <semaphore.h>
sem_t sem;
sem_init(&sem, 0, 3);   // 3 accès simultanés max
sem_wait(&sem);         // P() — décrémenter
sem_post(&sem);         // V() — incrémenter
Sans mutex : compteur++ = 3 instructions ASM → race condition garantie.
Deadlock : T1 attend verrou de T2 qui attend verrou de T1 → toujours verrouiller dans le même ordre.

⑥ Anti-patterns & Pièges courants

JS Callback Hell — imbrication illimitée. Fix : async/await.
JS await en for…of — séquentiel ! Fix : Promise.all(arr.map(f)).
JS Promise non gérée — UnhandledRejection. Toujours .catch() ou try/catch.
Py GIL oublié — threading ne parallélise pas le CPU. Fix : multiprocessing.
Py time.sleep() dans async — bloque l'event loop. Fix : await asyncio.sleep().
Py threading.Lock() oublié — variable partagée corrompue. Toujours protéger.
C Race condition — accès concurrent sans mutex → UB. Toujours mutex_lock.
C Deadlock — T1 attend T2 qui attend T1. Fix : toujours verrouiller dans le même ordre.
C Thread leak — thread jamais joint ni détaché. Fix : pthread_join() ou pthread_detach().
∀ Variable locale partagée — passer des pointeurs vers des locales détruites après return = UB.
